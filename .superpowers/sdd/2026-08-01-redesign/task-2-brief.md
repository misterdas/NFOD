function parseCSV(csv) {
  const lines = csv.trim().split("\n");
  if (!lines.length) return [];
  const headers = lines[0].split(",").map(s => s.trim().replace(/^"|"$/g, ""));
  const rows = [];
  for (let i = 1; i < lines.length; i++) {
    if (!lines[i].trim()) continue;
    const cells = lines[i].split(",").map(s => s.trim().replace(/^"|"$/g, ""));
    const row = {};
    headers.forEach((h, j) => {
      let val = cells[j];
      if (h !== "Client Type" && h !== "Date" && val !== undefined) val = parseFloat(val) || 0;
      row[h] = val;
    });
    rows.push(row);
  }
  return rows;
}

window.NFOD = window.NFOD || {};
NFOD.data = (function () {
  const availableDates = NFOD.utils.sortDatesChronological(
    Array.from(new Set(rawCSVData.map(r => r.Date)))
  );
  function getParticipantMap(date) {
    const map = {};
    rawCSVData.forEach(r => { if (r.Date === date) map[r["Client Type"]] = r; });
    return map;
  }
  async function loadMoneyFlow() {
    try {
      const res = await fetch(NFOD.utils.cacheBust("docs/money_flow_data.json"));
      if (!res.ok) return null;
      return await res.json();
    } catch (e) { return null; }
  }
  async function loadOHLC() {
    try {
      const res = await fetch(NFOD.utils.cacheBust("docs/ohlc_data.json"));
      if (!res.ok) return null;
      return await res.json();
    } catch (e) { return null; }
  }
  return { availableDates, getParticipantMap, loadMoneyFlow, loadOHLC, parseCSV, rawCSVData };
